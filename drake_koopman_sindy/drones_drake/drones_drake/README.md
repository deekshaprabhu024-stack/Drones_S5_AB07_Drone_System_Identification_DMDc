## Run order (do this exactly in this order)

```bash
cd AB07_contribution

# 1. Validate the pipeline on a known nonlinear system first (safety net)
python3 identification/lotka_volterra.py

# 2. Generate the meshgrid vector-field comparison for Lotka-Volterra
#    (works immediately; the Drake part of this script needs step 3 first)
python3 identification/meshgrid_viz.py

# 3. Run the Drake circle-tracking simulation (opens Meshcat in your browser --
#    open the printed URL to watch the drone fly live)
python3 drake_sim/circle_tracking.py

# 4. Run EDMDc and SINDy on the Drake data, compare against plain DMDc
python3 identification/compare_methods.py

# 5. Run the Kalman filter on noisy Drake data
python3 identification/kalman_filter.py

# 6. (optional, now that Drake data exists) re-run meshgrid_viz.py to also
#    generate the Drake roll-pitch vector field
python3 identification/meshgrid_viz.py
```
